import React, { createContext, useContext, useState, useEffect } from "react";
import type { LoginCredentials, ApiError, AuthResponse } from "../types";
import { authService } from "../services/authService";
import {
  saveSession,
  loadSession,
  clearSession,
} from "../services/tokenStorage";
import { setSessionExpiredHandler } from "../services/authEvents";
import { ROLE_NAMES } from "../types/role";
import { getEmployeeIdFromToken } from "../services/decodeToken";
import { scheduleService } from "@/features/employeeSchedules/services/scheduleService";

interface AuthContextValue {
  user: AuthResponse | null;
  loading: boolean;
  initializing: boolean;
  error: string | null;
  errorCode: string | null;
  pendingIdentifier: string;
  handleLogin: (identifier: string, password: string) => Promise<void>;
  handleLogout: () => Promise<void>;
  clearError: () => void;
}
const ALLOWED_MOBILE_ROLES: string[] = [
  ROLE_NAMES.Cliente,
  ROLE_NAMES.Mesero,
  ROLE_NAMES.Cajero,
  ROLE_NAMES.Repartidor,
  ROLE_NAMES.Cocinero,
  ROLE_NAMES.AuxiliarCocina,
];
const AuthContext = createContext<AuthContextValue | undefined>(undefined);

export const AuthProvider: React.FC<{ children: React.ReactNode }> = ({
  children,
}) => {
  const [user, setUser] = useState<AuthResponse | null>(null);
  const [loading, setLoading] = useState(false);
  const [initializing, setInitializing] = useState(true);
  const [error, setError] = useState<string | null>(null);
  const [errorCode, setErrorCode] = useState<string | null>(null);
  const [pendingIdentifier, setPendingIdentifier] = useState("");

    useEffect(() => {
        (async () => {
        const session = await loadSession();
        if (session) {
          setUser(session);
          // El hasActiveShift/activeShiftRoleName guardado es del momento del
          // login — puede estar desactualizado si el turno cambió desde
          // entonces. Lo revalidamos contra el backend antes de resolver
          // a qué panel entra el empleado.
          await refreshShiftStatus(session);
        }
        setInitializing(false);
        })();
    }, []);

    const refreshShiftStatus = async (session: AuthResponse) => {
      const employeeId = getEmployeeIdFromToken(session.token);
      if (!employeeId) return; // Cliente/Gerente/Administrador sin EmployeeId: no aplica

      const shift = await scheduleService.getCurrentShift(employeeId);
      if (!shift) return; // falló la consulta: seguimos con el valor cacheado

      const updated: AuthResponse = {
        ...session,
        hasActiveShift: shift.hasActiveShift,
        activeShiftRoleName: shift.roleName,
      };

      setUser(updated);
      await saveSession(updated);
    };
    useEffect(() => {
      setSessionExpiredHandler(async () => {
        await clearSession();
        setUser(null);
      });
    }, []);
  const clearError = () => {
    setError(null);
    setErrorCode(null);
  };

  const handleLogin = async (identifier: string, password: string) => {
    setError(null);
    setErrorCode(null);

    if (!identifier || !password) {
      setError("Por favor ingresa tu documento/correo y contraseña");
      return;
    }

    setLoading(true);
    try {
      const credentials: LoginCredentials = { identifier, password };
      const response = await authService.login(credentials);

const tieneAccesoMovil = response.roles.some((r) =>
  ALLOWED_MOBILE_ROLES.includes(r),
);

      if (!tieneAccesoMovil) {
        setError(
          "Esta cuenta no tiene acceso a la app móvil. Usa el panel web de SaborExpress.",
        );
        setLoading(false);
        return;
      }

      await saveSession(response);
      setUser(response);
    } catch (err) {
      const apiError = err as ApiError;
      setError(apiError.message ?? "Ocurrió un error al iniciar sesión");
      setErrorCode(apiError.code ?? null);
      setPendingIdentifier(identifier);
    } finally {
      setLoading(false);
    }
  };

  const handleLogout = async () => {
    try {
      await authService.logout();
    } finally {
      await clearSession();
      setUser(null);
    }
  };

  return (
    <AuthContext.Provider
      value={{
        user,
        loading,
        initializing,
        error,
        errorCode,
        pendingIdentifier,
        handleLogin,
        handleLogout,
        clearError,
      }}
    >
      {children}
    </AuthContext.Provider>
  );
};

export const useAuth = () => {
  const ctx = useContext(AuthContext);
  if (!ctx) throw new Error("useAuth debe usarse dentro de un AuthProvider");
  return ctx;
};

import {
  getToken,
  getRefreshToken,
  saveSession,
  clearSession,
} from "./tokenStorage";
import { AUTH_ENDPOINTS } from "./authEndpoints";
import { notifySessionExpired } from "./authEvents";

// Todos los que pidan renovar el token (fetchAuthed, SignalR) comparten UNA sola llamada.
let refreshPromise: Promise<string | null> | null = null;

// Reintenta solo ante fallas de red reales (tunnel no listo, wifi cortado, etc.)
// No reintenta ante 401/500/etc., esos ya son respuestas válidas del servidor.
async function fetchWithRetry(
  input: RequestInfo,
  init: RequestInit,
  retries = 2,
  delayMs = 1000,
): Promise<Response> {
  try {
    return await fetch(input, init);
  } catch (err) {
    const isNetworkError =
      err instanceof TypeError &&
      /failed to fetch|network request failed/i.test(err.message);

    if (retries > 0 && isNetworkError) {
      await new Promise((res) => setTimeout(res, delayMs));
      return fetchWithRetry(input, init, retries - 1, delayMs * 2);
    }
    throw err;
  }
}

export function refreshAccessToken(): Promise<string | null> {
  if (!refreshPromise) {
    refreshPromise = doRefresh().finally(() => {
      refreshPromise = null;
    });
  }
  return refreshPromise;
}

async function doRefresh(): Promise<string | null> {
  const refreshToken = await getRefreshToken();
  if (!refreshToken) return null;

  try {
    const res = await fetchWithRetry(AUTH_ENDPOINTS.refreshToken, {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        Accept: "*/*",
        "X-Tunnel-Skip-AntiPhishing-Page": "true",
      },
      body: JSON.stringify({ refreshToken }),
    });

    if (!res.ok) return null;

    const data = await res.json();

    // Guardamos la sesión completa renovada (nuevo token + nuevo refreshToken rotado)
    await saveSession({
      token: data.token,
      refreshToken: data.refreshToken,
      identifier: data.identifier,
      roles: data.roles ?? [],
      hasActiveShift: data.hasActiveShift ?? true,
      activeShiftRoleName: data.activeShiftRoleName ?? null,
    });

    return data.token;
  } catch {
    return null;
  }
}

interface ApiFetchOptions extends RequestInit {
  skipAuth?: boolean; // para endpoints públicos (login, register, etc.)
}

export async function fetchAuthed(
  url: string,
  options: ApiFetchOptions = {},
): Promise<Response> {
  const { skipAuth, headers, ...rest } = options;

  const token = skipAuth ? null : await getToken();

  const finalHeaders = {
    "Content-Type": "application/json",
    Accept: "*/*",
    // Evita la página de advertencia ("Continue anyway") de Dev Tunnels
    "X-Tunnel-Skip-AntiPhishing-Page": "true",
    ...(token ? { Authorization: `Bearer ${token}` } : {}),
    ...headers,
  };

  let res = await fetchWithRetry(url, { ...rest, headers: finalHeaders });

  // Si da 401 y no es un endpoint público, intentamos renovar UNA vez
  if (res.status === 401 && !skipAuth) {
    const newToken = await refreshAccessToken();

    if (newToken) {
      res = await fetchWithRetry(url, {
        ...rest,
        headers: { ...finalHeaders, Authorization: `Bearer ${newToken}` },
      });
    } else {
      // El refresh también falló: la sesión ya no es válida de verdad
      await clearSession();
      notifySessionExpired();
    }
  }

  return res;
}

import type { ApiError } from "../types";

export async function handleResponse<T>(res: Response): Promise<T> {
  if (!res.ok) {
    const body = await res.json().catch(() => null);

    const errorData: ApiError = {
      message: body?.error ?? body?.message ?? "Error de red o del servidor",
      statusCode: res.status,
      code: body?.code,
    };

    throw errorData;
  }
  return res.json();
}

import { Platform } from "react-native";
import * as SecureStore from "expo-secure-store";
import type { AuthResponse } from "../types";

const TOKEN_KEY = "saborexpress_token";
const REFRESH_KEY = "saborexpress_refresh_token";
const USER_KEY = "saborexpress_user";

const isWeb = Platform.OS === "web";

async function setItem(key: string, value: string): Promise<void> {
  if (isWeb) {
    localStorage.setItem(key, value);
  } else {
    await SecureStore.setItemAsync(key, value);
  }
}

async function getItem(key: string): Promise<string | null> {
  if (isWeb) {
    return localStorage.getItem(key);
  }
  return SecureStore.getItemAsync(key);
}

async function deleteItem(key: string): Promise<void> {
  if (isWeb) {
    localStorage.removeItem(key);
  } else {
    await SecureStore.deleteItemAsync(key);
  }
}

export async function saveSession(session: AuthResponse) {
  await setItem(TOKEN_KEY, session.token);
  await setItem(REFRESH_KEY, session.refreshToken ?? "");
  await setItem(USER_KEY, JSON.stringify(session));
}

export async function loadSession(): Promise<AuthResponse | null> {
  const userJson = await getItem(USER_KEY);
  if (!userJson) return null;

  try {
    const parsed = JSON.parse(userJson);
    // Validamos que tenga la forma real esperada; si no, la sesión es inválida.
    if (!parsed?.identifier || !Array.isArray(parsed?.roles)) {
      await clearSession();
      return null;
    }
    return parsed as AuthResponse;
  } catch {
    await clearSession();
    return null;
  }
}

export async function clearSession() {
  await deleteItem(TOKEN_KEY);
  await deleteItem(REFRESH_KEY);
  await deleteItem(USER_KEY);
}

export async function getToken(): Promise<string | null> {
  return getItem(TOKEN_KEY);
}

export async function getRefreshToken(): Promise<string | null> {
  return getItem(REFRESH_KEY);
}

export interface DecodedToken {
  nameid: string;
  unique_name: string;
  role: string;
  EmployeeId?: string;
  BranchId?: string;
  CustomerId?: string;
  exp: number;
  iat: number;
}

export function decodeToken(token: string): DecodedToken | null {
  try {
    const payload = token.split(".")[1];
    // atob funciona en web; en nativo React Native también lo soporta desde RN 0.7x+ / Hermes moderno
    const decoded = JSON.parse(atob(payload));
    return decoded;
  } catch {
    return null;
  }
}

export function getEmployeeIdFromToken(token: string): number | null {
  const decoded = decodeToken(token);
  if (!decoded?.EmployeeId) return null;
  return Number(decoded.EmployeeId);
}

export function getBranchIdFromToken(token: string): number | null {
  const decoded = decodeToken(token);
  if (!decoded?.BranchId) return null;
  return Number(decoded.BranchId);
}

export function getCustomerIdFromToken(token: string): number | null {
  const decoded = decodeToken(token);
  if (!decoded?.CustomerId) return null;
  return Number(decoded.CustomerId);
}

// El id de Usuario (no el de Empleado) viene en el claim estándar "nameid"
// (ClaimTypes.NameIdentifier). Es el mismo Id que usa el backend para
// agrupar las notificaciones por usuario en el Hub de SignalR.
export function getUserIdFromToken(token: string): number | null {

  // El Auxiliar de Cocina entra al MISMO tablero que el Cocinero, en modo
  // solo lectura (ChefOrders.tsx ya oculta los botones de avanzar estado).
  [ROLE_NAMES.AuxiliarCocina]: "/chef",
  // Gerente y Administrador todavia no tienen pantalla movil dedicada;
  // ajustar cuando exista (probablemente panel web, no app movil)
};

export const ROLE_LABELS: Record<RoleName, string> = {
  [ROLE_NAMES.Cliente]: "CLIENTE",
  [ROLE_NAMES.Repartidor]: "REPARTIDOR",
  [ROLE_NAMES.Cocinero]: "COCINERO",
  [ROLE_NAMES.Mesero]: "MESERO",
  [ROLE_NAMES.Cajero]: "CAJERO",
  [ROLE_NAMES.AuxiliarCocina]: "AUXILIAR DE COCINA",

};

// Un usuario puede tener varios roles (ej. Cliente + Mesero).
// Prioriza el primer rol OPERATIVO que tenga; si no tiene ninguno, usa Cliente.
const PRIORITY_ORDER: RoleName[] = [

  ROLE_NAMES.Mesero,
  ROLE_NAMES.Cajero,
  ROLE_NAMES.Cocinero,
  ROLE_NAMES.AuxiliarCocina,
  ROLE_NAMES.Repartidor,
  ROLE_NAMES.Cliente,
];

export function resolvePrimaryRole(roles: string[]): RoleName {
  for (const candidate of PRIORITY_ORDER) {
    if (roles.includes(candidate)) return candidate;
  }
  return ROLE_NAMES.Cliente;
}

// Si el rol es operativo (Mesero, Cajero, Cocinero, Repartidor) pero no
// tiene turno activo ahora mismo, lo tratamos como Cliente para efectos
// de a qué panel entra — sigue siendo su rol real, solo cambia a dónde navega.
export function resolveEffectiveRole(
  roles: string[],
  hasActiveShift: boolean,
  activeShiftRoleName?: string | null
): RoleName {
  // Si el backend ya nos dice EXACTAMENTE cuál rol tiene el turno activo,
  // usamos ese directo — sin adivinar por prioridad fija. Esto resuelve
  // el caso de un empleado con más de un rol operativo (ej. Mesero +
  // Repartidor), donde antes siempre ganaba el de mayor prioridad aunque
  // el turno real de hoy fuera otro.
  if (activeShiftRoleName && roles.includes(activeShiftRoleName)) {
    return activeShiftRoleName as RoleName;
  }

  const primary = resolvePrimaryRole(roles);

  if (primary !== ROLE_NAMES.Cliente && !hasActiveShift) {
    return ROLE_NAMES.Cliente;
  }

  return primary;
}

=================== features\notifications\context\UserNotificationsContext.tsx ===================
import React, {
  createContext,
  useCallback,
  useContext,
  useEffect,
  useMemo,
  useRef,
  useState,
} from "react";
import { AppState, StyleSheet, View } from "react-native";
import { useSafeAreaInsets } from "react-native-safe-area-context";
import * as signalR from "@microsoft/signalr";
import { useAudioPlayer, setAudioModeAsync } from "expo-audio";
import { API_BASE_URL } from "@/config/api";
import { useAuth } from "@/features/auth/context/AuthContext";
import { getToken } from "@/features/auth/services/tokenStorage";
import { refreshAccessToken } from "@/features/auth/services/apiClient";
import {
  decodeToken,
  getUserIdFromToken,
} from "@/features/auth/services/decodeToken";
import { normalizeNotificationType } from "../utils/notificationType";
import { notificationService } from "../services/notificationService";
import {
  NotificationToastContainer,
  type ToastNotification,
} from "@/features/shared/components/NotificationToast";
import { AppAlertModal } from "@/features/shared/components/AppAlertModal";

const notifySound = require("../../../assets/sounds/notify.wav");

setAudioModeAsync({ playsInSilentMode: true }).catch(() => {});

// Tipos que hacen sonar el celular del cliente / repartidor.
const SOUND_TYPES = new Set([
  "PaymentConfirmed",
  "DeliveryAssigned",
  "OrderStatusChanged",
  "OrderReady",
]);

type Listener = (n: ToastNotification) => void;

interface UserNotificationsValue {
  history: ToastNotification[];
  unreadCount: number;
  markAllRead: () => void;
  clear: () => void;
  subscribe: (fn: Listener) => () => void;
}

const Ctx = createContext<UserNotificationsValue | undefined>(undefined);

export function UserNotificationsProvider({
  children,
}: {
  children: React.ReactNode;
}) {
  const { user } = useAuth();
  const insets = useSafeAreaInsets();
  const player = useAudioPlayer(notifySound);
  const playerRef = useRef(player);
  playerRef.current = player;

  const [history, setHistory] = useState<ToastNotification[]>([]);
  const [toasts, setToasts] = useState<ToastNotification[]>([]);
  const [unreadCount, setUnreadCount] = useState(0);
  const historyRef = useRef<ToastNotification[]>([]);
  const listenersRef = useRef(new Set<Listener>());

  const userId = user?.token ? getUserIdFromToken(user.token) : null;

  useEffect(() => {
    historyRef.current = history;
  }, [history]);

  const emit = useCallback((n: ToastNotification) => {
    listenersRef.current.forEach((fn) => {
      try {
        fn(n);
      } catch {
        // un listener roto no debe afectar a los demás
      }
    });
  }, []);

  // Carga el historial del servidor (también sirve para recuperar lo que llegó
  // mientras la app estaba en segundo plano o sin conexión).
  const loadHistory = useCallback(async () => {
    if (!userId) return;
    try {
      const list = await notificationService.getByUser(userId);
      const mapped: ToastNotification[] = list.map((n) => ({
        id: n.id,
        title: n.title,
        message: n.message,
        type: normalizeNotificationType(n.type),
        relatedEntityType: n.relatedEntityType,
        relatedEntityId: n.relatedEntityId,
      }));
      setHistory(mapped.slice(0, 50));
      setUnreadCount(list.filter((n) => !n.isRead).length);
    } catch {
      // silencioso: se reintenta en la próxima reconexión
    }
  }, [userId]);

  useEffect(() => {
    loadHistory();
  }, [loadHistory]);

  useEffect(() => {
    if (!user?.token) return;

    const connection = new signalR.HubConnectionBuilder()
      .withUrl(`${API_BASE_URL}/hubs/notifications`, {
        accessTokenFactory: async () => {
          const token = await getToken();
          if (!token) return "";
          const decoded = decodeToken(token);
          const expiring = !decoded || decoded.exp * 1000 < Date.now() + 10000;
          if (expiring) {
            const refreshed = await refreshAccessToken();
            return refreshed ?? "";
          }
          return token;
        },
        withCredentials: false,
        headers: { "X-Tunnel-Skip-AntiPhishing-Page": "true" },
      })
      .withAutomaticReconnect({
        nextRetryDelayInMilliseconds: ({ previousRetryCount }) => {
          if (previousRetryCount === 0) return 0;
          if (previousRetryCount === 1) return 2000;
          if (previousRetryCount === 2) return 10000;
          return 30000;
        },
      })
      .build();

    connection.on("ReceiveNotification", (raw: any) => {
      const n: ToastNotification = {
        ...raw,
        type: normalizeNotificationType(raw.type),
      };

      const alreadyKnown = historyRef.current.some((p) => p.id === n.id);
      if (!alreadyKnown) {
        setHistory((prev) => [n, ...prev].slice(0, 50));
        setUnreadCount((c) => c + 1);
      }
      setToasts((prev) => [...prev, n]);

      if (SOUND_TYPES.has(n.type)) {
        try {
          playerRef.current.seekTo(0);
          playerRef.current.play();
        } catch {
          // un fallo de audio no debe romper la notificación
        }
      }
      emit(n);
    });

    // Al reconectar, se recupera lo que pudo perderse y se avisa a las
    // pantallas para que refresquen sus datos ("Resync").
    const resync = () => {
      loadHistory();
      emit({ id: 0, title: "", message: "", type: "Resync" });
    };
    connection.onreconnected(resync);

    let disposed = false;
    let retryTimer: ReturnType<typeof setTimeout> | null = null;

    const startIfDown = () => {
      if (disposed) return;
      if (connection.state !== signalR.HubConnectionState.Disconnected) return;
      connection
        .start()
        .then(resync)
        .catch(() => {
          // Falló el arranque (401, túnel caído...): reintenta en 5 s
          if (!disposed) retryTimer = setTimeout(startIfDown, 5000);
        });
    };
    startIfDown();

    // Al volver a primer plano: reconecta si el SO mató el socket y refresca.
    const appStateSub = AppState.addEventListener("change", (state) => {
      if (state === "active") {
        if (connection.state === signalR.HubConnectionState.Disconnected) {
          startIfDown();
        } else {
          resync();
        }
      }
    });

    return () => {
      disposed = true;
      if (retryTimer) clearTimeout(retryTimer);
      appStateSub.remove();
      connection.stop();
    };
  }, [user?.token, loadHistory, emit]);

  const subscribe = useCallback((fn: Listener) => {
    listenersRef.current.add(fn);
    return () => {
      listenersRef.current.delete(fn);
    };
  }, []);

  const markAllRead = useCallback(() => {
    setUnreadCount(0);
    if (userId) notificationService.markAllAsRead(userId).catch(() => {});
  }, [userId]);

  const clear = useCallback(() => {
    const ids = historyRef.current.map((n) => n.id).filter((id) => id > 0);
    setHistory([]);
    setUnreadCount(0);
    ids.forEach((id) => notificationService.delete(id).catch(() => {}));
  }, []);

  const dismissToast = useCallback((id: number) => {
    setToasts((prev) => prev.filter((t) => t.id !== id));
  }, []);

  const value = useMemo(
    () => ({ history, unreadCount, markAllRead, clear, subscribe }),
    [history, unreadCount, markAllRead, clear, subscribe],
  );

  return (
    <Ctx.Provider value={value}>
      <View style={{ flex: 1 }}>
        {children}
        <View
          pointerEvents="box-none"
          style={[StyleSheet.absoluteFill, { paddingTop: insets.top }]}
        >
          <NotificationToastContainer
            notifications={toasts}
            onDismiss={dismissToast}
          />
        </View>
        {/* Necesario para el botón "Limpiar" de la campana (showAppAlert) */}
        <AppAlertModal />
      </View>
    </Ctx.Provider>
  );
}

export function useUserNotifications() {
  const ctx = useContext(Ctx);
  if (!ctx)
    throw new Error(
      "useUserNotifications debe usarse dentro de UserNotificationsProvider",
    );
  return ctx;
}

// Ejecuta `cb` cada vez que llega una notificación (o "Resync" tras reconectar).
export function useNotificationListener(cb: Listener) {
  const { subscribe } = useUserNotifications();
  const ref = useRef(cb);
  ref.current = cb;
  useEffect(() => subscribe((n) => ref.current(n)), [subscribe]);
}


=================== features\shared\hooks\useKitchenNotifications.ts ===================
import { useEffect, useRef, useState } from "react";
import { Platform } from "react-native";
import * as signalR from "@microsoft/signalr";
import { useAudioPlayer, setAudioModeAsync } from "expo-audio";
import { API_BASE_URL } from "@/config/api";
import { useAuth } from "@/features/auth/context/AuthContext";
import { normalizeNotificationType } from "@/features/notifications/utils/notificationType";
import { getToken } from "@/features/auth/services/tokenStorage";
import { refreshAccessToken } from "@/features/auth/services/apiClient";
import { decodeToken } from "@/features/auth/services/decodeToken";

const notifySound = require("../../../assets/sounds/notify.wav");
const kitchenNotifySound = require("../../../assets/sounds/kitchen-notify.wav");

// Por defecto, iOS silencia cualquier audio de la app si el switch lateral
// del celular está en modo silencioso. Esto lo desactiva para que el sonido
// de "pedido nuevo/listo" siempre suene, sin importar esa posición del switch.
// En Android no hace falta, pero no afecta si se llama igual.
setAudioModeAsync({
  playsInSilentMode: true,
}).catch(() => {
  // Si falla (por ejemplo en web, donde no aplica), no debe romper nada.
});

interface KitchenNotification {
  id: number;
  title: string;
  message: string;
  type: string | number;
  relatedEntityType?: string | null;
  relatedEntityId?: number | null;
}

// Conecta al Hub de notificaciones en tiempo real y reproduce un sonido
// cada vez que llega una notificación mientras la app está en primer plano.
export function useKitchenNotifications(
  onNotification?: (n: KitchenNotification) => void,
) {
  const { user } = useAuth();
  const connectionRef = useRef<signalR.HubConnection | null>(null);
  // Dos sonidos distintos para que mesero y cocinero no los confundan:
  // OrderReady (mesero) usa notify.wav, OrderCreated (cocinero) usa el otro.
  const waiterPlayer = useAudioPlayer(notifySound);
  const kitchenPlayer = useAudioPlayer(kitchenNotifySound);

  // En web, los navegadores bloquean cualquier audio que no haya sido
  // disparado directamente por un gesto del usuario (click/tap). Por eso
  // arrancamos "bloqueados" en web hasta que el usuario toque el aviso, y
  // ese primer toque reproduce el sonido una vez para "desbloquearlo" — de
  // ahí en adelante ya puede sonar solo el resto de la sesión. En apps
  // nativas (Android/iOS) esta restricción no existe, así que ahí arrancamos
  // ya habilitados.
  const [soundEnabled, setSoundEnabled] = useState(Platform.OS !== "web");

  // El listener de SignalR se registra UNA sola vez (el useEffect de abajo
  // solo depende de user?.token) y queda con el valor de soundEnabled
  // "congelado" del momento en que se registró. Usamos un ref para que el
  // listener siempre lea el valor MÁS RECIENTE, no el de cuando se conectó.
  const soundEnabledRef = useRef(soundEnabled);
  useEffect(() => {
    soundEnabledRef.current = soundEnabled;
  }, [soundEnabled]);

  const enableSound = () => {
    try {
      // Desbloqueamos ambos players con este primer gesto del usuario,
      // así no importa cuál de los dos suene primero después.
      waiterPlayer.seekTo(0);
      waiterPlayer.play();
      kitchenPlayer.seekTo(0);
      kitchenPlayer.play();
      setSoundEnabled(true);
    } catch (e) {
      console.log("[DEBUG] No se pudo desbloquear el sonido:", e);
    }
  };

  useEffect(() => {
    if (!user?.token) return;

    const connection = new signalR.HubConnectionBuilder()
    .withUrl(`${API_BASE_URL}/hubs/notifications`, {
        // Lee el token MÁS RECIENTE en cada conexión/reconexión, en vez de
        // capturar user.token una sola vez al montar el efecto. Así, si el
        // token expiró (15 min) y ya fue refrescado en segundo plano por
        // apiClient.ts, SignalR usa el nuevo token al reconectar, en vez
        // de seguir mandando uno vencido y recibir 401 para siempre.
        accessTokenFactory: async () => {
          const token = await getToken();
          if (!token) return "";

          // Si el token ya expiró (o expira en menos de 10s), lo refrescamos
          // ANTES de que SignalR lo use, en vez de mandar uno vencido y
          // recibir 401 sin que nada dispare el refresh (SignalR no pasa
          // por fetchAuthed, así que este chequeo es necesario aquí también).
          const decoded = decodeToken(token);
          const isExpiredOrExpiring =
            !decoded || decoded.exp * 1000 < Date.now() + 10000;

          if (isExpiredOrExpiring) {
            const refreshed = await refreshAccessToken();
            return refreshed ?? "";
          }

          return token;
        },
        withCredentials: false, // usamos Bearer token, no cookies del navegador
        // Evita la página de advertencia de Dev Tunnels tanto en el
        // negotiate (fetch) como en el WebSocket/long-polling posterior.
        headers: {
          "X-Tunnel-Skip-AntiPhishing-Page": "true",
        },
    })
    // Por defecto, withAutomaticReconnect() solo reintenta 4 veces
    // (0s, 2s, 10s, 30s) y luego se rinde para siempre — el usuario tendría
    // que recargar la página manualmente para volver a conectar. Con esta
    // política personalizada, reintenta INDEFINIDAMENTE: espera creciente
    // hasta un tope de 30s entre intentos, así que si el túnel de Dev
    // Tunnels vuelve después de estar caído 5 minutos, la app se reconecta
    // sola sin que nadie tenga que hacer nada.
    .withAutomaticReconnect({
      nextRetryDelayInMilliseconds: (retryContext) => {
        const { previousRetryCount } = retryContext;
        if (previousRetryCount === 0) return 0;
        if (previousRetryCount === 1) return 2000;
        if (previousRetryCount === 2) return 10000;
        return 30000; // desde el 4to intento en adelante, cada 30s, sin límite
      },
    })
    .build();

    connection.on(
      "ReceiveNotification",
      (notification: KitchenNotification) => {
        const normalizedType = normalizeNotificationType(notification.type);

        // Suena cuando el pedido está LISTO (le llega a Mesero/Cajero) o
        // cuando llega un pedido NUEVO (le llega a Cocinero). El backend ya
        // filtra por rol al elegir a quién mandar cada tipo de notificación,
        // así que aquí no hace falta distinguir por rol: si a este usuario
        // le llegó, es porque le corresponde sonar.
        if (soundEnabledRef.current) {
          const playerToUse =
            normalizedType === "OrderReady"
              ? waiterPlayer
              : normalizedType === "OrderCreated"
                ? kitchenPlayer
                : null;

          if (playerToUse) {
            try {
              playerToUse.seekTo(0);
              playerToUse.play();
            } catch {
              // Si el sonido falla (por ejemplo, silencio del dispositivo),
              // no debe romper la recepción de la notificación.
            }
          }
        }
        // Mandamos el type ya normalizado para que toasts / historial /
        // íconos también funcionen sin importar cómo serializó el backend.
        onNotification?.({ ...notification, type: normalizedType });
      },
    );

  connection.start().catch(async (err) => {
    const tokenDebug = await getToken();
    console.log("Error conectando al Hub de notificaciones:", err);
    console.log(
      "[DEBUG] Token usado:",
      tokenDebug ? tokenDebug.slice(0, 20) + "..." : "NULO/VACÍO",
    );
    console.log("[DEBUG] URL del hub:", `${API_BASE_URL}/hubs/notifications`);
  });
    connectionRef.current = connection;

    return () => {
      connection.stop();
      connectionRef.current = null;
    };
  }, [user?.token]);

  return { soundEnabled, enableSound };
}
